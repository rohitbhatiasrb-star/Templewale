function openTemple(name, location){ document.getElementById('modalTempleName').textContent = name; document.getElementById('modalTempleLocation').textContent = location; document.getElementById('templeModal').classList.add('show'); } function closeTemple(e){ if( e.target.id === 'templeModal' || e.target.classList.contains('close-modal') ){ document.getElementById('templeModal').classList.remove('show'); } }
